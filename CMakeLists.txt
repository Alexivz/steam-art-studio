cmake_minimum_required(VERSION 3.22)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" STUDIO_VERSION LIMIT_COUNT 1)
string(REGEX MATCH "^[0-9]+\\.[0-9]+\\.[0-9]+" STUDIO_NUMERIC_VERSION "${STUDIO_VERSION}")
project(SteamArtStudio VERSION "${STUDIO_NUMERIC_VERSION}" LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(GNUInstallDirs)
include(CTest)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Widgets Network Concurrent Svg)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(Qt6 6.2 QUIET COMPONENTS DBus)
endif()

add_library(studio_core STATIC
  src/valvekv.cpp src/steam.cpp src/cache.cpp src/network.cpp src/transactions.cpp
  src/libraryindex.cpp src/draft.cpp src/transfers.cpp src/animation.cpp)
target_include_directories(studio_core PUBLIC src)
target_link_libraries(studio_core PUBLIC Qt6::Core Qt6::Gui Qt6::Network)
add_library(studio_ui STATIC src/models.cpp src/preview.cpp src/window.cpp src/credentials.cpp src/ui.cpp src/workspace.cpp)
target_link_libraries(studio_ui PUBLIC studio_core Qt6::Widgets Qt6::Concurrent Qt6::Svg)
if(TARGET Qt6::DBus)
  target_link_libraries(studio_ui PRIVATE Qt6::DBus)
endif()
if(MSVC)
  add_compile_options(/utf-8 /W4)
  target_compile_options(studio_core PUBLIC /utf-8)
else()
  target_compile_options(studio_core PRIVATE -Wall -Wextra)
  target_compile_options(studio_ui PRIVATE -Wall -Wextra)
endif()
if(WIN32)
  target_compile_definitions(studio_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
qt_add_executable(steam-art-studio src/main.cpp assets/resources.qrc)
target_link_libraries(steam-art-studio PRIVATE studio_ui)
set_target_properties(steam-art-studio PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" WIN32_EXECUTABLE TRUE)
if(WIN32)
  enable_language(RC)
  configure_file(packaging/windows/app.rc.in "${CMAKE_BINARY_DIR}/studio.rc" @ONLY)
  target_sources(steam-art-studio PRIVATE "${CMAKE_BINARY_DIR}/studio.rc")
  set(STUDIO_INSTALL_BIN ".")
  set(STUDIO_INSTALL_DOC "docs")
else()
  set(STUDIO_INSTALL_BIN "${CMAKE_INSTALL_BINDIR}")
  set(STUDIO_INSTALL_DOC "${CMAKE_INSTALL_DATADIR}/doc/steam-art-studio")
endif()
install(TARGETS steam-art-studio RUNTIME DESTINATION "${STUDIO_INSTALL_BIN}")
install(FILES LICENSE THIRD_PARTY_NOTICES.md README.md VERSION CHANGELOG.md DESTINATION "${STUDIO_INSTALL_DOC}")
set(STUDIO_BUNDLED_FFMPEG_DIR "" CACHE PATH "Optional verified FFmpeg prefix with its notices and corresponding source")
if(STUDIO_BUNDLED_FFMPEG_DIR)
  if(WIN32)
    set(STUDIO_FFMPEG_NAME "ffmpeg.exe")
    set(STUDIO_FFMPEG_DEST ".")
  else()
    set(STUDIO_FFMPEG_NAME "ffmpeg")
    set(STUDIO_FFMPEG_DEST "${CMAKE_INSTALL_LIBEXECDIR}/steam-art-studio")
  endif()
  if(NOT EXISTS "${STUDIO_BUNDLED_FFMPEG_DIR}/bin/${STUDIO_FFMPEG_NAME}" OR NOT EXISTS "${STUDIO_BUNDLED_FFMPEG_DIR}/share/doc/studio-ffmpeg/build.json")
    message(FATAL_ERROR "The FFmpeg prefix must contain the executable, build record, licenses and matching source")
  endif()
  install(PROGRAMS "${STUDIO_BUNDLED_FFMPEG_DIR}/bin/${STUDIO_FFMPEG_NAME}" DESTINATION "${STUDIO_FFMPEG_DEST}")
  install(DIRECTORY "${STUDIO_BUNDLED_FFMPEG_DIR}/share/doc/studio-ffmpeg/" DESTINATION "${STUDIO_INSTALL_DOC}/ffmpeg")
endif()
install(DIRECTORY docs/ DESTINATION "${STUDIO_INSTALL_DOC}/guides" FILES_MATCHING PATTERN "*.md" PATTERN "HISTORIAL*" EXCLUDE)
if(UNIX AND NOT APPLE)
  install(FILES packaging/linux/org.steamartstudio.SteamArtStudio.desktop DESTINATION "${CMAKE_INSTALL_DATADIR}/applications")
  install(FILES assets/steam-art-studio.svg DESTINATION "${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps" RENAME org.steamartstudio.SteamArtStudio.svg)
  install(FILES packaging/linux/org.steamartstudio.SteamArtStudio.metainfo.xml DESTINATION "${CMAKE_INSTALL_DATADIR}/metainfo")
endif()
if(WIN32 AND NOT CMAKE_CROSSCOMPILING)
  find_program(STUDIO_WINDEPLOYQT windeployqt HINTS "${Qt6_DIR}/../../../bin" REQUIRED)
  configure_file(packaging/windows/deploy.cmake.in "${CMAKE_BINARY_DIR}/deploy-windows.cmake" @ONLY)
  install(SCRIPT "${CMAKE_BINARY_DIR}/deploy-windows.cmake")
endif()

if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND AND NOT CMAKE_CROSSCOMPILING)
    add_test(NAME studio-packaging-tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_packaging.py")
  endif()
  find_package(Qt6 REQUIRED COMPONENTS Test)
  function(studio_test name source output)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE studio_core Qt6::Test)
    set_target_properties(${name} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/${output}/bin")
    add_test(NAME ${name} COMMAND ${name} -txt)
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 180)
  endfunction()
  studio_test(studio-portable-tests tests/test_portable.cpp build-portable-tests)
  studio_test(studio-playback-tests tests/test_playback.cpp build-playback-tests)
  target_link_libraries(studio-playback-tests PRIVATE studio_ui)
  if(UNIX AND NOT APPLE)
    # Existing process-fault fixtures intentionally remain Linux-only.
    add_executable(studio-crash-worker tests/crash_worker.cpp)
    target_link_libraries(studio-crash-worker PRIVATE studio_core)
    set_target_properties(studio-crash-worker PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/build-crash-tests/bin")
    studio_test(studio-tests tests/test_studio.cpp build-tests)
    add_dependencies(studio-tests studio-crash-worker)
    studio_test(studio-network-tests tests/test_network.cpp build-network-tests)
    studio_test(studio-gui-tests tests/test_gui.cpp build-gui-tests)
    target_link_libraries(studio-gui-tests PRIVATE studio_ui)
    target_sources(studio-gui-tests PRIVATE assets/resources.qrc)
    configure_file(VERSION "${CMAKE_BINARY_DIR}/VERSION" COPYONLY)
  endif()
endif()
include(packaging/CPackOptions.cmake)
